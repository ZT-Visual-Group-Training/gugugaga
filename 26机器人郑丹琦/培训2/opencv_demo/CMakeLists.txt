cmake_minimum_required(VERSION 3.20)
project(opencv_demo LANGUAGES CXX)
set(OpenCV_DIR "C:/opencv/build")
find_package(OpenCV REQUIRED)
message(STATUS "OpenCV found: ${OpenCV_VERSION}")
include_directories(${OpenCV_INCLUDE_DIRS})
add_executable(opencv_demo src/main.cpp)
target_link_libraries(opencv_demo ${OpenCV_LIBS})
add_custom_command(TARGET opencv_demo POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${OpenCV_DIR}/x64/vc16/bin/opencv_world4140.dll ${OpenCV_DIR}/x64/vc16/bin/opencv_world4140d.dll $<TARGET_FILE_DIR:opencv_demo>)
